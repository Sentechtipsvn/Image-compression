  function fail(reason, retryable = false) {
    requestVersion += 1;

  function loadQr() {

    cleanupRequest = () => {
      clearTimeout(timer);
      image.removeEventListener('load', onLoad);
      image.removeEventListener('error', onError);
    };
    image.addEventListener('load', onLoad);
    image.addEventListener('error', onError);
    image.src = activeSource;
  }

  function optionalText(value, key) {
    if (value === undefined || value === null) return '';
    if (typeof value !== 'string') throw new Error('Khóa ' + key + ' cần kiểu Văn bản.');
    return value.trim();
  }

  function normalize(data) {
    if (!data || typeof data !== 'object' || Array.isArray(data)) {
      throw new Error('Cần một Từ điển có các khóa money, number, name và bank.');
    }
    const name = optionalText(data.name, 'name');
    if (!name) throw new Error('Chưa có tên người nhận trong khóa name.');
    if (typeof data.number !== 'string') {
      throw new Error('Đặt khóa number thành kiểu Văn bản để giữ số 0 đầu tài khoản.');
    }
    const number = data.number.trim();
    if (!/^[A-Za-z0-9]{1,19}$/.test(number)) {
      throw new Error('Khóa number cần số tài khoản hoặc biệt danh gồm chữ/số, tối đa 19 ký tự.');
    }
    if (typeof data.bank !== 'string' && typeof data.bank !== 'number') {
      throw new Error('Khóa bank cần mã BIN ngân hàng gồm 6 chữ số.');
    }
    const bank = String(data.bank).trim();
    if (!/^\d{6}$/.test(bank)) throw new Error('Khóa bank cần mã BIN ngân hàng gồm 6 chữ số.');

    let money = null;
    const value = typeof data.money === 'string' ? data.money.trim() : data.money;
    if (value !== undefined && value !== null && value !== '') {
      if ((typeof value !== 'number' && typeof value !== 'string') ||
          (typeof value === 'number' && !Number.isSafeInteger(value))) {
        throw new Error('Khóa money cần số nguyên, ví dụ 125000.');
      }
      const digits = String(value).trim();
      if (!/^\d{1,13}$/.test(digits) || Number(digits) <= 0) {
        throw new Error('Khóa money cần số nguyên dương tối đa 13 chữ số, không có dấu chấm hoặc dấu phẩy.');
      }
      money = Number(digits);
    }
    const note = optionalText(data.note, 'note');
    if (note.length > 50) throw new Error('Khóa note cần tối đa 50 ký tự.');
    const bankName = optionalText(data.bankName, 'bankName');
    const qr = optionalText(data.qr, 'qr');
    return { name, number, bank, money, note, qr, bankName };
  }

  function qrSource(payment) {
    if (payment.qr) {
      const source = payment.qr.replace(/\s+/g, '');
      if (/^data:image\/(?:png|jpeg|webp|svg\+xml);base64,[A-Za-z0-9+/]+={0,2}$/i.test(source)) {
        return source;
      }
      if (/^[A-Za-z0-9+/]+={0,2}$/.test(source)) {
        if (source.startsWith('iVBORw0KGgo')) return 'data:image/png;base64,' + source;
        if (source.startsWith('/9j/')) return 'data:image/jpeg;base64,' + source;
      }
      throw new Error('Khóa qr cần Base64 của ảnh PNG/JPEG hoặc Data URI ảnh hợp lệ.');
    }
    const url = new URL('https://img.vietqr.io/image/' + payment.bank + '-' + payment.number + '-qr_only.svg');
    if (payment.money !== null) url.searchParams.set('amount', String(payment.money));
    if (payment.note) url.searchParams.set('addInfo', payment.note);
    return url.href;
  }

  function renderBankName(name) {
    const target = byId('bank-value');
    const letters = typeof Intl.Segmenter === 'function'
      ? Array.from(new Intl.Segmenter('vi', { granularity: 'grapheme' }).segment(name), item => item.segment)
      : Array.from(name.normalize('NFC'));
    const cycle = Math.max(2.6, ((letters.length - 1) * .085 + .7) / .74);
    const fragment = document.createDocumentFragment();
    letters.forEach((letter, index) => {

    });
    target.setAttribute('aria-label', name);
    target.style.setProperty('--bank-wave-cycle', cycle.toFixed(3) + 's');
    target.replaceChildren(fragment);
  }
